\begin{theorem}
The sum of two even integers is even.
\end{theorem}
\begin{proof}
Write the integers as \(2a\) and \(2b\). Their sum is \(2(a+b)\).
\end{proof}
\begin{lemma}

\end{lemma}
\begin{proposition}
The sum of two odd integers is even.
\end{proposition}
